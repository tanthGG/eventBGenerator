MACHINE uM2
REFINES pM1
SEES cM2

VARIABLES
  pktFwdr
  pktData
  createdPkts
  waitingBuff
  sentDown
  sentUp
  destBuff 
  recvBuff
  clrRecvBuffFlg
  ctlNeighbours 
  

EVENTS
  Initialisation
    extends
    begin
      @int01 destBuff := ∅
      @int02 waitingBuff := ∅
      @int03 pktFwdr := ∅
      @int04 pktData := ∅
      @int05 createdPkts := ∅
      @int06 sentUp := ∅
      @int07 sentDown := ∅
      @int08 ctlNeighbours := ∅
      @int09 recvBuff := ∅
      @int10 clrRecvBuffFlg := ∅
    end

Event creatingDataPacket ≙
refines creatingPkt

any
  x
  des
  pkt
  data
where
  @g01 x ∈ ND \ Dests 
  @g02 pkt ∈ PKT 
  @g03 data ∈ ℤ
  @g04 x = initialSrc(pkt) 
  @g05 des = ran({pkt} ◁ finalDest)
  @g06 pkt ∉ dom(pktFwdr)
  @g07 pkt ∉ dom(pktData)
  @g08 x ↦ pkt ∉ waitingBuff
  @g09 type(pkt) = DATA
then
  @a01 createdPkts := createdPkts ∪ {pkt}
  @a02 pktFwdr := pktFwdr ∪ {pkt ↦ x}
  @a03 pktData := pktData ∪ {pkt ↦ data}
  @a04 waitingBuff := waitingBuff ∪ {x ↦ pkt}
end

Event creatingControlPacket ≙
refines creatingPkt

any
  x
  des
  pkt
  data
where
  @g01 x ∈ ND \ Dests 
  @g02 pkt ∈ PKT 
  @g03 data ∈ ℤ
  @g04 x = initialSrc(pkt) 
  @g05 des = ran({pkt} ◁ finalDest)
  @g06 pkt ∉ dom(pktFwdr)
  @g07 pkt ∉ dom(pktData)
  @g08 x ↦ pkt ∉ waitingBuff
  @g09 data = CTL_VAL
  @g10 type(pkt) ∈ CONTROL
then
  @a01 createdPkts := createdPkts ∪ {pkt}
  @a02 pktFwdr := pktFwdr ∪ {pkt ↦ x}
  @a03 pktData := pktData ∪ {pkt ↦ data}
  @a04 waitingBuff := waitingBuff ∪ {x ↦ pkt}
end

Event start_tx ≙
extends start_tx

any
  x
  pkt
where
  @g01 x ∈ ND 
  @g02 pkt ∈ PKT
  @g03 x ↦ pkt ∉ sentDown
  @g04 x ↦ pkt ∈ waitingBuff
  @g05 pkt ∈ dom(pktFwdr)
then
  @a01 sentDown := sentDown ∪ {x ↦ pkt}
  @a02 waitingBuff := waitingBuff \ {x ↦ pkt}
  @a03 pktFwdr := pktFwdr <+ {pkt ↦ x} 
end

Event send_down ≙
extends send_down

any
  x
  pkt
where
  @g01 x ∈ ND 
  @g02 pkt ∈ PKT
  @g03 x ↦ pkt ∈ sentDown
then
  skip
end

Event send_up ≙
extends send_up

any
  x
  pkt
  nbrs
where
  @g01 nbrs ∈ {n ∣ n ∈ ℙ(ND)} 
  @g02 x ∈ ND 
  @g03 pkt ∈ PKT
  @g04 nbrs ≠ ∅
  @g05 pkt ∉ dom(ctlNeighbours)
  @g06 x ↦ pkt ∉ sentUp
  @g07 x ↦ pkt ∈ sentDown
  
then
  @a01 ctlNeighbours := ctlNeighbours ∪ ({pkt} × nbrs)
  @a02 sentDown := sentDown \ {x ↦ pkt}
  @a03 sentUp := sentUp ∪ {x ↦ pkt}
end

Event receive ≙
extends receive

any
  x
  pkt
where
  @g01 x ∈ ND 
  @g02 pkt ∈ PKT
  @g03 x ↦ pkt ∉ recvBuff
  @g04 x ↦ pkt ∉ sentUp
  @g05 x ↦ pkt ∉ sentDown
  @g06 pkt ↦ x ∉ ctlNeighbours
then
  @a01 recvBuff := recvBuff ∪ {x ↦ pkt}
  @a02 ctlNeighbours := ctlNeighbours \ {pkt ↦ x}
end

Event clear_recvdBuff ≙
extends clear_recvdBuff

any
  x
  pkt
where
  @g01 : x ∈ ND 
  @g02 pkt ∈ PKT
  @g03 x ↦ pkt ∈ recvBuff
  @g04 x ↦ pkt ∈ clrRecvBuffFlg
then
  @a01 recvBuff := recvBuff \ {x ↦ pkt}
end

Event fwdr_receive_pkt ≙
extends fwdr_receive_pkt

any
  x
  pkt
where
  @g01 x ∈ ND 
  @g02 pkt ∈ PKT
  @g03 x ↦ pkt ∈ recvBuff
  @g04 x ↦ pkt ∉ clrRecvBuffFlg
  @g05 x ∉ Dests
  @g06 x ↦ pkt ∉ waitingBuff
then
  @a01 clrRecvBuffFlg := clrRecvBuffFlg ∪ {x ↦ pkt}
  @a02 waitingBuff := waitingBuff ∪ {x ↦ pkt}
end

Event dest_recv_pkt ≙
extends dest_recv_pkt

any
  nb
  pkt
where
  @g01 nb ∈ ND 
  @g02 pkt ∈ PKT
  @g03 nb ↦ pkt ∈ recvBuff
  @g04 nb ∈ Dests
  @g05 nb ↦ pkt ∉ clrRecvBuffFlg
  @g06 nb ↦ pkt ∉ destBuff
then
  @a01 clrRecvBuffFlg := clrRecvBuffFlg ∪ {nb ↦ pkt}
  @a02 destBuff := destBuff ∪ {nb ↦ pkt}
  
end

Event finish_tx_pkt ≙
extends finish_tx_pkt

any
  x
  pkt
where
  @g01 x ∈ ND 
  @g02 pkt ∈ PKT
  @g03 {pkt} ◁ ctlNeighbours = ∅
  @g04 pkt ∈ ran(sentUp) 
  @g05 pkt ∉ ran(sentDown)
  @g06 x ∈ dom(sentUp)
  @g07 x ↦ pkt ∈ sentUp
then
  @a01 sentUp := sentUp \ {x ↦ pkt}
end

END
